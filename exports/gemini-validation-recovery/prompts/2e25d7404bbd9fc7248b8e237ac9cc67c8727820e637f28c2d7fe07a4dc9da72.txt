    sid = streamid_from_addr(addr);

    addr >>= 2;

    switch (addr) {

            /* Tailptr mode is always on.  */

            value |= DMACR_TAILPTR_MODE;

            s->regs[addr] = value;

                stream_reset(s);

            }

            break;

            value &= ~(value & DMASR_IRQ_MASK);

    }

    stream_update_irq(s);
