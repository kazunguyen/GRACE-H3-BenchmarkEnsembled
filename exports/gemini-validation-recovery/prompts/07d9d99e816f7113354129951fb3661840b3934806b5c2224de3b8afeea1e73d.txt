        goto fail;

    }

    s->state = 1;



    /* Open the backing file */

                          flags, true, false, &local_err);

    if (ret < 0) {

        error_propagate(errp, local_err);

        goto fail;

    }



    /* Set request alignment */

    align = qemu_opt_get_size(opts, "align", bs->request_alignment);
